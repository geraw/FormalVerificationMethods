\question{התאמת תכונות להרכבה}{
נתונות שתי מערכות מעברים והקבוצה $H=\{\langle a,b\rangle\}$. בחרו את הטענה הנכונה על ההרכבה לפי כללי זוג פעולות.

\begin{center}
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[font=\bfseries] at (0,1.5) {$TS_1$};
  \node[st] (p0) at (-1,0.5) {$p_0$};
  \node[st] (p1) at (1,0.5) {$p_1$};
  \draw[arr] (-1.8,0.5) -- (p0);
  \draw[arr] (p0) -- node[lab,above] {$a$} (p1);
  \draw[arr] (p1) to[loop right] node[lab] {$c$} (p1);

  \node[font=\bfseries] at (5,1.5) {$TS_2$};
  \node[st] (q0) at (4,0.5) {$q_0$};
  \node[st] (q1) at (6,0.5) {$q_1$};
  \draw[arr] (3.2,0.5) -- (q0);
  \draw[arr] (q0) -- node[lab,above] {$b$} (q1);
  \draw[arr] (q1) to[loop right] node[lab] {$d$} (q1);
\end{tikzpicture}
\end{center}

\answerfour
  {מן המצב $\langle p_0,q_0\rangle$ יש צעד מסונכרן אל $\langle p_1,q_1\rangle$ עם התווית $\langle a,b\rangle$.}
  {מן המצב $\langle p_0,q_0\rangle$ חייבים לבצע קודם צעד אינטרליבינג על $a$ בלבד.}
  {ההרכבה ריקה כי $a$ ו-$b$ אינן אותה פעולה.}
  {בכל מצב חייבים לסנכרן גם את $c$ עם $d$, אף ש-$\langle c,d\rangle\notin H$.}
}
