\tikzset{
  nodebox/.style={draw=black, rounded corners=4pt, very thick, fill=gray!4,
    minimum width=8mm, minimum height=6mm, align=center, inner sep=1.5pt},

  labelbox/.style={inner sep=3pt, font=\scriptsize, align=center},

\newcommand{\initarrow}[2]{\draw[initial,auto] #1 -- node[labelbox,auto] {$x{=}0\land y{=}0\land z{=}0$} (#2);}
\newcommand{\selfloop}[3]{\draw[edge,loop above,looseness=7] (#1) to node[labelbox] {#2} (#3);}
\newcommand{\graphframe}[1]{
  \draw[graphframe] (-3.65,3.15) rectangle (6.65,-3.75);
  \node[title, anchor=west, fill=white, inner sep=2pt] at (-3.35,2.85) {#1};
}

\newcommand{\correctgraph}[1]{
  \graphframe{#1}
  \node[nodebox] (l0) at (0,0) {$\ell_0$};
  \node[nodebox] (l1) at (5,0) {$\ell_1$};
  \node[nodebox] (l2) at (4,-3) {$\ell_2$};
  \node[nodebox] (l3) at (-1,-1.5) {$\ell_3$};
  \node[finalnode] (lend) at (-2,-3) {$\ell_4$};
  \initarrow{(-3.1,0)}{l0.west}
  \draw[edge] (l0) -- node[labelbox] {$x<2:x:=x+1$} (l1);
  \draw[edge,bend left=14] (l1) to node[labelbox] {$y=0:y:=1$} (l0);
  \draw[edge] (l1) -- node[labelbox] {$y=1:y:=0$} (l2);
  \draw[edge] (l2) -- node[labelbox] {$\mathit{true}:z:=z+1$} (l0);
  \draw[edge,loop above,looseness=7] (l0) to node[labelbox] {$x=2:y:=1;x:=x+1$} (l0);
  \draw[edge] (l0) -- node[labelbox] {$x>2:\mathrm{skip}$} (l3);
  \draw[edge] (l3) -- node[labelbox] {$\mathit{true}:z:=z+1$} (lend);
}
