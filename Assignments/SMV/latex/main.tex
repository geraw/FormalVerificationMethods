\section{הנחיות הגשה}
יש להגיש קובץ דחוס אחד, המכיל \textbf{בדיוק} את ארבעת הקבצים הבאים:
\begin{itemize}
  \item \texttt{q1.smv}
  \item \texttt{q2.smv}
  \item \texttt{q3.smv}
  \item \texttt{answers.txt}
\end{itemize}

\noindent
הקובץ \texttt{answers.txt} חייב להיכתב \textbf{בדיוק} לפי התבנית שבסוף מסמך זה.

\subsection*{כללים מחייבים}
\begin{enumerate}
  \item בכל אחד מן הקבצים \texttt{q1.smv}, \texttt{q2.smv}, \texttt{q3.smv} חייב להיות בדיוק מודול עליון אחד בשם \texttt{main}\RLM.
  \item כאשר שאלה מגדירה במפורש שמות של משתנים, טיפוסים, או תחומי ערכים, אין לשנות אותם.
  \item מותר להוסיף \texttt{DEFINE} כרצונכם. משתני מצב או קלט נוספים מותרים רק אם השאלה אינה מקבעת ממשק חובה.
  \item כל מעבר חייב להיות מוגדר בצורה מלאה; אין להשאיר משתנה ללא הגדרת \texttt{next}\RLM.
  \item בכל שאלה יש לבדוק בדיוק את התכונות שנדרשו, ולא תכונות אחרות.
  \item ריצה, עקבת נגד או עדות נגישות שתוגש ב-\texttt{answers.txt} חייבת לציין \textbf{את כל המשתנים} של המודל בכל מצב.
\end{enumerate}

\section{שאלה 1: מציאת תקלה בפרוטוקול שגוי למניעה הדדית}
להלן פסאודו־קוד של שני תהליכים זהים, \texttt{P0} ו-\texttt{P1}, המשתמשים בשני דגלים משותפים:

\begin{englishlisting}
\begin{lstlisting}
shared boolean want0 := false;
shared boolean want1 := false;

process Pi (i in {0,1}, j = 1-i):
loop forever:
  L0: wait until !wantj;
  L1: wanti := true;
  L2: critical section;
  L3: wanti := false;
end loop
\end{lstlisting}
\end{englishlisting}

\subsection*{הנחות מידול מחייבות}
\begin{itemize}
  \item המערכת פועלת בשזירה: בכל צעד מתקדם תהליך אחד בלבד.
  \item כל שורה בפסאודו־קוד \textenglish{(L0, L1, L2, L3)} מייצגת צעד אטומי אחד.
  \item בפרט, בדיקת התנאי ב-\textenglish{L0} והתוצאה שלה (להישאר ב-\textenglish{L0} או לעבור ל-\textenglish{L1}) מתבצעות באותו צעד אטומי.
  \item הקטע הקריטי (L2) חייב להיות מיוצג במודל כמצב בקרה נפרד.
  \item בתחילת הריצה שני הדגלים הם \texttt{false}, ושני התהליכים מתחילים ב-\textenglish{L0}\RLM.
\end{itemize}

\subsection*{ממשק חובה}
על המודל להגדיר (דרך \texttt{VAR} או \texttt{DEFINE}) שני משתנים בוליאניים:
\begin{itemize}
  \item \texttt{in\_crit0} -- אמת אם ורק אם תהליך \texttt{P0} נמצא בקטע הקריטי.
  \item \texttt{in\_crit1} -- אמת אם ורק אם תהליך \texttt{P1} נמצא בקטע הקריטי.
\end{itemize}

\subsection*{התכונה שיש לבדוק}
יש להוסיף למודל \textbf{בדיוק} את תכונת הבטיחות הבאה:
\begin{englishlisting}
\begin{lstlisting}
INVARSPEC !(in_crit0 & in_crit1)
\end{lstlisting}
\end{englishlisting}

\subsection*{מה מותר ומה אסור}
\begin{itemize}
  \item חובה להשתמש בשמות \texttt{in\_crit0} ו-\texttt{in\_crit1} עבור הממשק.
  \item מותר לבחור כל שמות משתנים אחרים וכל קידוד סביר למצבי הבקרה.
  \item מותר להוסיף משתנה עזר לצורך שזירה, למשל משתנה בורר תהליך.
  \item אסור לשנות את התנהגות הפסאודו־קוד.
  \item אסור לפצל שורה אחת לשני צעדים אטומיים.
\end{itemize}

\subsection*{מה יש לענות}
ב-\texttt{answers.txt} יש למלא:
\begin{itemize}
  \item האם התכונה נכונה או שגויה.\RLM
  \item אם היא שגויה: אורך ריצת הנגד הקצרה ביותר.\RLM
  \item אם היא שגויה: ריצת הנגד הקצרה ביותר, כשרשומים בכל מצב כל משתני המודל.\RLM
  \item משפט קצר המסביר מהו הבאג ברמת האלגוריתם.\RLM
\end{itemize}

\end{lstlisting}
\end{englishlisting}

\begin{itemize}

  \item יש להוסיף למודל \textbf{בדיוק} את תכונת הבטיחות: \textenglish{INVARSPEC x >= 0}\RLM
  \item ב-\texttt{answers.txt} יש לציין את ערכי כל המשתנים במודל שבניתם (כולל משתני בקרה ודגלים מקומיים)\RLM

\subsection*{מה יש לענות}
ב-\texttt{answers.txt} יש למלא:
\begin{itemize}

  \item אורך הריצה הקצרה ביותר שמגיעה לראשונה לערך \texttt{v}.\RLM

\section{שאלת בונוס: מציאת באג תזמון multi-threading במאגר חסום}
שאלה זו היא שאלת בונוס. מטרתה אינה לתרגם קוד שורה־אחר־שורה ל-\textenglish{SMV}, אלא למצוא הפשטה סופית מתאימה לקוד \textenglish{multi-threaded}, כך שניתן יהיה לחשוף בעזרת בודק המודלים באג אמיתי בתזמון.

להלן קוד דמוי-\textenglish{Java} של מאגר חסום בגודל \textenglish{N=1}\RLM. שני \textenglish{producer threads} מריצים שוב ושוב את \texttt{put}, ו-\textenglish{consumer thread} אחד מריץ שוב ושוב את \texttt{take}\RLM.

\begin{englishlisting}
\begin{lstlisting}
class BoundedBuffer {
    private final int N = 1;
    private int count = 0;

    synchronized void put() throws InterruptedException {
        if (count == N) {
            wait();
        }

        count++;

        notify();
    }

    synchronized void take() throws InterruptedException {
        if (count == 0) {
            wait();
        }

        count--;

        notify();
    }
}
\end{lstlisting}
\end{englishlisting}

הכוונה של הקוד היא לשמור תמיד על התנאי:
\begin{englishlisting}
\begin{lstlisting}
0 <= count <= N
\end{lstlisting}
\end{englishlisting}

\subsection*{משימת המידול}
בנו מודל \textenglish{SMV} סופי, בקובץ \texttt{q4.smv}, המייצג את ההתנהגות הרלוונטית של הקוד תחת שזירה של \textenglish{threads}\RLM.
אין צורך, ואף אין רצון, למדל את כל שפת \textenglish{Java}\RLM. עליכם לבחור הפשטה מספיק עשירה כדי ללכוד את הבאג.

המודל חייב לכלול לפחות את הרעיונות הבאים:
\begin{itemize}
  \item מצב בקרה נפרד לכל \textenglish{thread}, למשל מיקום נוכחי בתוך \texttt{put} או \texttt{take}\RLM.
  \item משתנה משותף \texttt{count}\RLM.
  \item ייצוג של המנעול \textenglish{(monitor lock)}, כלומר מי מחזיק כרגע את המנעול או שהמנעול פנוי.
  \item ייצוג של \textenglish{threads} הממתינים אחרי קריאה ל-\texttt{wait}\RLM.
  \item מתזמן לא-דטרמיניסטי שבכל צעד מקדם לכל היותר \textenglish{thread} אחד.
  \item קריאה ל-\texttt{notify} המעירה באופן לא-דטרמיניסטי לכל היותר \textenglish{thread} ממתין אחד.
\end{itemize}

\subsection*{סמנטיקה חשובה של \texttt{wait}}
יש למדל את \texttt{wait} בהתאם למשמעות המקובלת של \textenglish{Java monitors}:
\begin{itemize}
  \item \textenglish{thread} שקורא ל-\texttt{wait} משחרר את המנעול ועובר למצב המתנה\RLM.
  \item לאחר \texttt{notify}, ה-\textenglish{thread} שהתעורר אינו ממשיך מיד בהכרח; הוא צריך לקבל שוב את המנעול\RLM.
  \item כאשר ה-\textenglish{thread} ממשיך אחרי \texttt{wait}, הוא ממשיך אחרי הקריאה ל-\texttt{wait}\RLM. בקוד הנתון אין בדיקה חוזרת של התנאי, כי השתמשו ב-\texttt{if} ולא ב-\texttt{while}\RLM.
\end{itemize}

\subsection*{ממשק חובה}
המודל חייב לכלול:
\begin{itemize}
  \item משתנה בשם \texttt{count} בטווח \textenglish{0..2}\RLM. הערך \texttt{2} מייצג גלישת קיבולת, כלומר מצב שבו \texttt{count > N}\RLM.
  \item משתנה בוליאני בשם \texttt{overflow}, שערכו אמת אם ורק אם בוצע \texttt{count++} כאשר \texttt{count = N}\RLM.
\end{itemize}

\subsection*{התכונה שיש לבדוק}
יש להוסיף למודל \textbf{בדיוק} את תכונת הבטיחות הבאה:
\begin{englishlisting}
\begin{lstlisting}
INVARSPEC !overflow
\end{lstlisting}
\end{englishlisting}

\subsection*{מה יש למצוא}
עליכם להשתמש ב-\textenglish{SMV} כדי למצוא ריצת נגד, אם קיימת, שבה יצרן ממשיך אחרי \texttt{wait} ומגדיל את \texttt{count}, למרות שבינתיים יצרן אחר כבר מילא את המאגר\RLM.

ריצת נגד טיפוסית, במילים, יכולה להיראות כך:
\begin{enumerate}
  \item המאגר מלא: \texttt{count = 1}\RLM.
  \item יצרן ראשון נכנס ל-\texttt{put}, מגלה שהמאגר מלא, וקורא ל-\texttt{wait}\RLM.
  \item הצרכן מוציא איבר, כך ש-\texttt{count} יורד ל-\texttt{0}, וקורא ל-\texttt{notify}\RLM.
  \item היצרן הראשון התעורר, אבל עדיין לא רץ\RLM.
  \item יצרן שני נכנס ל-\texttt{put} ומכניס איבר, כך ש-\texttt{count} חוזר ל-\texttt{1}\RLM.
  \item היצרן הראשון ממשיך אחרי \texttt{wait}\RLM. מכיוון שאין בדיקה חוזרת של התנאי, הוא מבצע \texttt{count++} וגורם לגלישה.
\end{enumerate}

\subsection*{מה יש לענות}
ב-\texttt{answers.txt} יש למלא:
\begin{itemize}
  \item האם התכונה \texttt{INVARSPEC !overflow} נכונה או שגויה\RLM.
  \item אורך ריצת הנגד הקצרה ביותר שמצאתם\RLM.
  \item ריצת הנגד, כאשר בכל מצב מופיעים כל משתני המודל שבניתם\RLM.
  \item הסבר קצר של הבאג במונחי הקוד המקורי\RLM.
  \item תיקון מוצע לקוד המקורי\RLM.
  \item הסבר קצר מדוע שימוש ב-\texttt{while} במקום \texttt{if} משנה את המודל ואת תוצאת הבדיקה\RLM.
\end{itemize}

\subsection*{רמז}
אין למדל את \texttt{wait} כאילו הוא ``ממתין עד שהתנאי נהיה נכון''. \texttt{wait} רק משחרר את המנעול וממתין להתראה. לאחר ההתעוררות, התנאי שבגללו ה-\textenglish{thread} המתין עלול להיות שוב שגוי\RLM.


\section{תבנית חובה לקובץ \texttt{answers.txt}}
יש להגיש את הקובץ \texttt{answers.txt} בדיוק בתבנית הבאה:

\begin{englishlisting}
\begin{lstlisting}
[Q1]
Verdict = TRUE/FALSE
ShortestCounterexampleLength = <integer or N/A>
State0 = (...)
State1 = (...)
...
BugExplanation = <one short sentence>

[Q2]
MinReachableX = <integer>
Invariant = TRUE/FALSE
ShortestWitnessLength = <integer>
State0 = (...)
State1 = (...)
...

[Q3]
Verdict = TRUE/FALSE
SmallestFailingPair = (a=<integer>, b=<integer>)
Bits = (a1=<0/1>, a0=<0/1>, b1=<0/1>, b0=<0/1>)
Expected = <integer>
Actual = <integer>
WrongLine = r1/r2/r3
Fix = <equation or N/A>

[Q4 - Bonus]
Verdict = TRUE/FALSE/NOT_SUBMITTED
ShortestCounterexampleLength = <integer or N/A>
State0 = (...)
State1 = (...)
...
BugExplanation = <one short sentence>
ProposedFix = <one short sentence>
WhyWhileHelps = <one short sentence>
\end{lstlisting}
\end{englishlisting}

\noindent
הערות:
\begin{itemize}
  \item אם בשאלה 1 התכונה נכונה, כתבו \texttt{ShortestCounterexampleLength = N/A} ואל תוסיפו מצבים.
  \item בשאלה 1, בכל מצב בריצת הנגד יש לציין את כל משתני המודל שבחרתם.
  \item אם בשאלה 3 התכונה נכונה, כתבו \texttt{Fix = N/A}\RLM.
  \item אם אינכם מגישים את שאלת הבונוס, כתבו בחלק של \textenglish{Q4} רק \texttt{Verdict = NOT\_SUBMITTED}, או השמיטו את החלק כולו לפי הנחיית המתרגל.
  \item בכל ריצה או עדות נגישות יש לציין את כל משתני המודל בכל מצב.
\end{itemize}
